57vector<string> tokenize(const string& expression) {
58 vector<string> tokens;
59 size_t i = 0;
60 while (i < expression.size()) {
61 //typecast into unsigned char
62 unsigned char ch = static_cast<unsigned char>(expression[i]);
63 // Ignore whitespace.
64 if (isspace(ch)) {
65 ++i;
66 continue;
67 }
68 // Recognize operators and parentheses.
69 if (string("()+*~>").find(expression[i])!= string::npos) {
70 tokens.push_back(string(1, expression[i]));
71 ++i;
72 continue;
73 }
74 // Recognize propositional atoms such as P, Q, x1, x12.
75 if (isalpha(ch) || ch == '_') {
76 size_t start = i;
77 while (i < expression.size()) {
78 unsigned char current =
79 static_cast<unsigned char>(expression[i]);
80
81 if (isalnum(current) || current == '_') {
82 ++i;
83 } else {
84 break;
85 }
86 }
87 tokens.push_back(
88 expression.substr(start, i - start)
89 );
90 continue;
91 }
92 throw invalid_argument(
93 "Invalid character in expression."
94 );
95 }
96 if (tokens.empty()) {
97 throw invalid_argument("The expression is empty.");
98 }
99 return tokens;
100}

102// Convert a fully parenthesized infix expression to prefix.
103string infixToPrefix(const string& expression) {
104 vector<string> tokens = tokenize(expression);
105 // Step 1: Reverse the tokens.
106 reverse(tokens.begin(), tokens.end());
107 // Step 2: Swap opening and closing parentheses.
108 for (string& token : tokens) {
109 if (token == "(") {
110 token = ")";
111 } else if (token == ")") {
112 token = "(";
113 }
114 }
115 // Step 3: Convert the modified expression to postfix.
116 vector<string> postfix;
117 stack<string> operators;
118 for (const string& token : tokens) {
119
120 // An operand goes directly to the output.
121 if (!isOperator(token) && token != "(" && token != ")") {
122 postfix.push_back(token);
123 }
124 // Opening parenthesis goes onto the stack.
125 else if (token == "(") {
126 operators.push(token);
127 }
128 // Pop operators until the matching opening parenthesis.
129 else if (token == ")") {
130 while (!operators.empty() && operators.top() != "(") {
131 postfix.push_back(operators.top());
132 operators.pop();
133 }
134 if (operators.empty()) {
135 throw invalid_argument(
136 "Mismatched parentheses."
137 );
138 }
139 operators.pop();
140 }
141 // Process logical operators.
142 else {
143 // Use strict greater-than because the original
144 // expression has been reversed.
145 while (!operators.empty() && operators.top() != "(" && precedence(operators.top()) > precedence(token)) {
146 postfix.push_back(operators.top());
147 operators.pop();
148 }
149 operators.push(token);
150 }
151 }
152 // Pop any remaining operators.
153 while (!operators.empty()) {
154 if (operators.top() == "(" ||
155 operators.top() == ")") {
156 throw invalid_argument(
157 "Mismatched parentheses."
158 );
159 }
160 postfix.push_back(operators.top());
161 operators.pop();
162 }
163 // Step 4: Reverse the postfix tokens to obtain prefix.
164 reverse(postfix.begin(), postfix.end());
165 // Step 5: Join the tokens with spaces.
166 string prefix;
167 for (const string& token : postfix) {
168 if (!prefix.empty()) {
169 prefix += ' ';
170 }
171 prefix += token;
172 }
173 return prefix;
174}

194//prefix to rooted binary parse tree
195Node* prefixToTree(const string& prefix) {
196 // Split the prefix expression into tokens.
197 istringstream input(prefix);
198 vector<string> tokens;
199 string token;
200
201 while (input >> token) {
202 tokens.push_back(token);
203 }
204 if (tokens.empty()) {
205 throw invalid_argument("Prefix expression is empty.");
206 }
207 // Stack stores roots of subtrees.
208 vector<Node*> st;
209 try {
210 // Process tokens from right to left.
211 for (auto it = tokens.rbegin(); it != tokens.rend(); ++it) {
212 token = *it;
213 if (token == "~") {
214 // Unary operator: requires one operand.
215 if (st.empty()) {
216 throw invalid_argument(
217 "Invalid prefix expression: missing operand for ~."
218 );
219 }
220 Node* child = st.back();
221 st.pop_back();
222 Node* node = new Node(token);
223 node->right = child;
224
225 st.push_back(node);
226 }
227 else if (token == "+" || token == "*" || token == ">") {
228 // Binary operator: requires two operands/subtrees.
229 if (st.size() < 2) {
230 throw invalid_argument(
231 "Invalid prefix expression: missing operands."
232 );
233 }
234 // IMPORTANT: first popped becomes LEFT child.
235 Node* leftChild = st.back();
236 st.pop_back();
237 Node* rightChild = st.back();
238 st.pop_back();
239
240 Node* node = new Node(token);
241 node->left = leftChild;
242 node->right = rightChild;
243
244 st.push_back(node);
245 }
246 else {
247 // Operand: create a leaf node.
248 if (!isValidAtomToken(token)) {
249 throw invalid_argument(
250 "Invalid token in prefix expression: " + token
251 );
252 }
253 st.push_back(new Node(token));
254 }
255 }
256 // A valid expression must produce exactly one root.
257 if (st.size() != 1) {
258 throw invalid_argument(
259 "Invalid prefix expression: multiple roots remain."
260 );
261 }
262 return st.back();
263 }
264 catch (...) {
265 // Clean up subtrees if construction fails.
266 for (Node* node : st) {
267 deleteTree(node);
268 }
269 throw;
270 }
271}

357
358 while (current != nullptr || !st.empty()) {

461// ---------- Iterative height ----------
462int heightIterative(Node* root) {
463 if (root == nullptr) {
464 return 0;
465 }
466
467 queue<Node*> q;
468 q.push(root);
469
470 int height = 0;
471
472 // Process the tree one level at a time.
473 while (!q.empty()) {
474 int nodesAtLevel = static_cast<int>(q.size());
475
476 // Process every node in the current level.
477 for (int i = 0; i < nodesAtLevel; i++) {
478 Node* current = q.front();
479 q.pop();

481 if (current->left != nullptr) {
482 q.push(current->left);

486 q.push(current->right);

488 }

490 height++;
491 }

551// ---------- Evaluate formula ----------
552bool evaluateFormula(
553 const Node* root,
554 const map<string, bool>& values
555) {
556 if (root == nullptr) {
557 throw invalid_argument(
558 "Cannot evaluate an empty parse tree."
559 );
560 }
561
562 const string& token = root->value;
563
564 // Negation: operand is stored in the RIGHT child.
565 if (token == "~") {
566 if (root->left != nullptr || root->right == nullptr) {
567 throw invalid_argument(
568 "Invalid parse tree: invalid operand structure for ~."
569 );
570 }
571
572 return !evaluateFormula(root->right, values);
573 }
574
575 // Binary operators.
576 if (token == "+" || token == "*" || token == ">") {
577 if (root->left == nullptr || root->right == nullptr) {
578 throw invalid_argument(
579 "Invalid parse tree: missing operand."
580 );
581 }
582
583 bool leftValue = evaluateFormula(root->left, values);
584 bool rightValue = evaluateFormula(root->right, values);
585
586 if (token == "+") {
587 return leftValue || rightValue;
588 }
589
590 if (token == "*") {
591 return leftValue && rightValue;
592 }
593
594 // Implication: P > Q means !P || Q.
595 return !leftValue || rightValue;
596 }
597
598 // Operand: return its assigned truth value.
599 auto it = values.find(token);
600
601 if (it == values.end()) {
602 throw invalid_argument(
603 "Truth value not provided for atom: " + token
604 );
605 }
606
607 return it->second;
608}

612void writeTruthTable(
613 const Node* root,
614 const string& filename
615) {
616 if (root == nullptr) {
617 throw invalid_argument(
618 "Cannot create a truth table for an empty tree."
619 );
620 }
621
622 vector<string> atoms = getAtoms(root);
623
624 if (atoms.empty()) {
625 throw invalid_argument(
626 "No atoms found in the parse tree."
627 );
628 }
629
630 // Truth-table generation is limited to 10 distinct atoms.
631 if (atoms.size() > 10) {
632 throw length_error(
633 "Truth table cannot be generated for more than "
634 "10 distinct atoms."
635 );
636 }
637
638 ofstream out(filename);
639
640 if (!out) {
641 throw runtime_error(
642 "Could not create output file: " + filename
643 );
644 }
645
646 out << "Truth Table (T = True, F = False)\n\n";
647
648 // Print column headings.
649 for (const string& atom : atoms) {
650 out << atom << '\t';
651 }
652
653 out << "Formula\n";
654
655 // Number of possible assignments = 2^n.
656 const unsigned int numberOfRows =
657 1U << atoms.size();
658
659 for (unsigned int row = 0;
660 row < numberOfRows;
661 row++) {
662
663 map<string, bool> assignment;
664
665 // Generate one assignment for every atom.
666 for (size_t i = 0; i < atoms.size(); i++) {
667 // The first atom changes slowest;
668 // the last atom changes fastest.
669 bool value =
670 ((row >> (atoms.size() - 1 - i)) & 1U) == 0;
671
672 assignment[atoms[i]] = value;
673
674 out << (value ? 'T' : 'F') << '\t';
675 }
676
677 // Evaluate the formula for this assignment.
678 bool formulaValue = evaluateFormula(root, assignment);
679
680 out << (formulaValue ? 'T' : 'F') << '\n';
681 }
682
683 if (!out) {
684 throw runtime_error(
685 "Error occurred while writing the truth table."
686 );
687 }
688}
